+import socket
+from typing import Optional, Callable, List, Dict
+import segno

+class LabelService:
+    def __init__(self, base_url: str, company: str, zpl_template: Optional[str] = None, connect: Optional[Callable] = None):
+        self.base_url = base_url
+        self.company = company
+        self.zpl_template = zpl_template or self._default_zpl_template()
+        self.connect = connect or socket.create_connection
+
+    def url_for(self, asset: Dict) -> str:
+        """Generate the URL for an asset."""
+        asset_tag = asset["asset_tag"]
+        return f"{self.base_url.rstrip('/')}/a/{asset_tag}"
+
+    def qr_svg(self, asset: Dict) -> str:
+        """Generate QR code SVG for an asset."""
+        url = self.url_for(asset)
+        qr_code = segno.make(url, error='m')
+        return qr_code.svg_inline(scale=4)
+
+    def zpl(self, asset: Dict) -> str:
+        """Generate ZPL label for an asset."""
+        url = self.url_for(asset)
+        return self.zpl_template.format(
+            asset_tag=asset["asset_tag"],
+            brand=asset["brand"],
+            model=asset["model"],
+            serial_number=asset["serial_number"],
+            url=url,
+            company=self.company
+        )
+
+    def _default_zpl_template(self) -> str:
+        """Return the default ZPL template."""
+        return ('^XA^CI28^PW400^LL200^FO20,20^BQN,2,4^FDQA,{url}^FS'
+                '^FO150,25^A0N,36,36^FD{asset_tag}^FS'
+                '^FO150,70^A0N,24,24^FD{brand} {model}^FS'
+                '^FO150,100^A0N,22,22^FDSN {serial_number}^FS'
+                '^FO150,140^A0N,20,20^FD{company}^FS^XZ')
+
+    def print_zpl(self, assets: List[Dict], host: str, port: int = 9100) -> int:
+        """Print ZPL labels to a network printer."""
+        count = 0
+        with self.connect(host, port, timeout=10) as sock:
+            for asset in assets:
+                zpl = self.zpl(asset)
+                sock.sendall(zpl.encode('utf-8'))
+                count += 1
+        return count
